class ItemProduk { public function __construct( public readonly string $sku, public readonly string $nama, public readonly float $harga, private int $stok, public readonly KategoriProduk $kategori ) {} public function kurangiStok(int $qty): void { if ($qty > $this->$stok) { throw new StokHabisException("Stok {$this->nama} tersisa {$this->stok}"); } $this->$stok -= $qty; } }
2️⃣ FIRST-CLASS COLLECTION

Koleksi Keranjang Belanja Berbasis SPL

Mengimplementasikan Countable & IteratorAggregate dengan pipeline fungsional modern.

Domain/Model/KeranjangCollection.php
<?php namespace App\Domain\Model; class KeranjangCollection implements \Countable, \IteratorAggregate { private array $items = []; public function tambah(ItemKeranjang $item): void { $this->$items[$item->produk->sku] = $item; } public function hitungSubtotal(): float { return array_reduce( $this->$items, fn($carry, $it) => $carry + $it->getSubtotal(), 0.0 ); } public function count(): int { return count($this->$items); } public function getIterator(): \Traversable { return new \ArrayIterator($this->$items); } }

5️⃣ PERSISTENSI FILE LOCK

Persistensi Transaksi Aman (LOCK_EX)

Menyimpan riwayat invoice ke berkas JSON dengan garansi integritas mutlak dari Race Condition.

Infrastructure/Persistence/JsonTransaksiRepository.php
public function simpan(Transaksi $trx): void { $fp = fopen($this->filePath, c+); if (!flock($fp, LOCK_EX)) { throw new \RuntimeException("Gagal mengunci transaksi database!"); } try { $isi = stream_get_contents($fp); $data = $isi ? json_decode($isi, true, flags: JSON_THROW_ON_ERROR) : []; $data[] = $trx->toArray(); ftruncate($fp, 0); rewind($fp); fwrite($fp, json_encode($data, JSON_PRETTY_PRINT | JSON_THROW_ON_ERROR)); } finally { flock($fp, LOCK_UN); // ✨ Garansi lepas kunci fclose($fp); } }
6️⃣ ORKESTRASI KASIR SERVICE

Orkestrator Bisnis: KasirService

Mengorkestrasi validasi stok, diskon, pembayaran, dan pencatatan dalam 1 transaksi atomik.

Application/Service/KasirService.php
public function checkout( KeranjangCollection $cart, DiskonStrategyInterface $diskon, PaymentProcessorInterface $payment, float $uangDiterima ): Transaksi { $subtotal = $cart->hitungSubtotal(); $potongan = $diskon->hitungPotongan($subtotal); $totalBayar = $subtotal - $potongan; $hasilBayar = $payment->proses($totalBayar, $uangDiterima); // Potong stok seluruh item produk foreach ($cart as $item) { $item->produk->kurangiStok($item->qty); $this->produkRepo->simpan($item->produk); } $trx = new Transaksi(TRX- . time(), $cart, $subtotal, $potongan, $totalBayar, $hasilBayar); $this->trxRepo->simpan($trx); return $trx; }
